cmake_minimum_required(VERSION 3.29)
project(vnote_host LANGUAGES C CXX)
include(GNUInstallDirs)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(VXM_BUILD_TESTS "Build real-core host boundary tests" OFF)
get_filename_component(VXM_REPOSITORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(VXM_CORE_ROOT "${VXM_REPOSITORY_ROOT}/third_party/vxcore")
set(VXCORE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(VXCORE_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(VXCORE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(VXCORE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(VXCORE_INSTALL OFF CACHE BOOL "" FORCE)
# Desktop assets are host-test harnesses, not a desktop VNote build. They need
# isolated mobile roots but must retain the ordinary desktop save implementation.
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(VXCORE_ENABLE_MOBILE_TEST_API ON CACHE BOOL "" FORCE)
else()
  set(VXCORE_ENABLE_MOBILE_TEST_API OFF CACHE BOOL "" FORCE)
endif()
if(ANDROID)
  if(ANDROID_PLATFORM_LEVEL LESS 28)
    message(FATAL_ERROR "vnote_host requires Android API 28 or newer (Bionic Iconv)")
  endif()
endif()
add_subdirectory("${VXM_CORE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/vxcore")

# Notices come from the exact built dependency sources, not a hand-maintained
# description of their licenses. Missing notices fail packaging.
set(NATIVE_NOTICES "VNote Mobile native host — GNU LGPL version 3\nApplication license texts are bundled separately.\n\n")
function(append_notice label path)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing native dependency license: ${label}: ${path}")
  endif()
  file(READ "${path}" text)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  string(APPEND NATIVE_NOTICES "\n===== ${label} =====\n${text}\n")
  set(NATIVE_NOTICES "${NATIVE_NOTICES}" PARENT_SCOPE)
endfunction()
append_notice("vxcore (MIT)" "${VXM_CORE_ROOT}/LICENSE")
append_notice("VNote cmark and bundled components" "${VXM_CORE_ROOT}/third_party/cmark/COPYING")
append_notice("libgit2 and bundled components; GPLv2 with linking exception" "${VXM_CORE_ROOT}/third_party/libgit2/COPYING")
get_target_property(sodium_source sodium SOURCE_DIR)
append_notice("libsodium CMake adapter" "${sodium_source}/LICENSE")
append_notice("libsodium" "${sodium_source}/libsodium/LICENSE")
get_target_property(pugi_notice pugixml VXCORE_LICENSE_FILE)
append_notice("pugixml" "${pugi_notice}")
if(NOT APPLE OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  get_target_property(curl_notice vxcore_webdav_curl VXCORE_LICENSE_FILE)
  append_notice("libcurl" "${curl_notice}")
endif()
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  if(NOT VXM_OPENSSL_LICENSE)
    message(FATAL_ERROR "The pinned OpenSSL source license must accompany target libraries")
  endif()
  append_notice("OpenSSL 3.5.9 (Apache-2.0)" "${VXM_OPENSSL_LICENSE}")
endif()
if(ANDROID)
  get_filename_component(ndk_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
  get_filename_component(ndk_prebuilt "${ndk_compiler_bin}/.." ABSOLUTE)
  append_notice("Android NDK LLVM runtime/toolchain notices" "${ndk_prebuilt}/NOTICE")
endif()
# The vendored JSON header retains SPDX attribution but not the complete MIT
# grant. Pin that grant to the matching upstream release and preserve attribution.
if(NOT VXM_JSON_LICENSE OR NOT EXISTS "${VXM_JSON_LICENSE}")
  message(FATAL_ERROR "Run scripts/native_deps.mjs and provide VXM_JSON_LICENSE")
endif()
file(SHA256 "${VXM_JSON_LICENSE}" json_license_hash)
if(NOT json_license_hash STREQUAL "86b998c792894ccb911a1cb7994f7a9652894e7a094c0b5e45be2f553f45cf14")
  message(FATAL_ERROR "Native JSON license hash mismatch")
endif()
append_notice("nlohmann/json 3.11.3 (MIT); header copyright 2013-2023 Niels Lohmann" "${VXM_JSON_LICENSE}")
string(APPEND NATIVE_NOTICES "\nSQLite: the authors disclaim copyright.\nMay you do good and not evil.\nMay you find forgiveness for yourself and forgive others.\nMay you share freely, never taking more than you give.\n")
# Hex bytes avoid raw-string delimiter collisions and MSVC's string-literal
# length limit. Storage is read-only and requires no startup allocation.
string(HEX "${NATIVE_NOTICES}" notice_hex)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," NATIVE_NOTICE_BYTES "${notice_hex}")
string(REGEX REPLACE "(0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,)" "\\1\n" NATIVE_NOTICE_BYTES "${NATIVE_NOTICE_BYTES}")
configure_file(licenses.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/licenses.cpp" @ONLY)

add_library(vnote_host SHARED vxcore_host.cpp "${CMAKE_CURRENT_BINARY_DIR}/licenses.cpp")
target_compile_definitions(vnote_host PRIVATE VXM_HOST_BUILD)
target_include_directories(vnote_host PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(vnote_host PRIVATE vxcore nlohmann_json sodium cmark)
set_target_properties(vnote_host PROPERTIES
  CXX_VISIBILITY_PRESET hidden
  C_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/artifacts"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/artifacts"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/artifacts"
  LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/artifacts")
if(ANDROID)
  target_link_options(vnote_host PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
if(MSVC)
  target_compile_options(vnote_host PRIVATE /W4 /utf-8)
else()
  target_compile_options(vnote_host PRIVATE -Wall -Wextra -Wpedantic)
endif()

  target_link_libraries(native_host_test PRIVATE vnote_host vxcore sqlite3)
  target_include_directories(native_host_test PRIVATE "${VXM_CORE_ROOT}/include")
  set_target_properties(native_host_test PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/artifacts"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/artifacts")
  if(MSVC)
    target_compile_options(native_host_test PRIVATE /utf-8)
  endif()
  add_test(NAME native_host_test COMMAND native_host_test)
endif()
